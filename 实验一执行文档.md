**HdfsApi.java**（公共工具类）
```java
package sdu.bigdata;

import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.*;

public class HdfsApi {
    private static final String HDFS_URI = "hdfs://localhost:19001";
    private static final Configuration conf = new Configuration();

    static { conf.set("fs.defaultFS", HDFS_URI); }

    public static FileSystem getFs() throws Exception {
        return FileSystem.get(conf);
    }
}
```
> 本机 `fs.defaultFS` 端口是 **19001**（不是默认 9000，WSL virtioproxy 对回环端口转发不稳定的历史沿革：9000→19000→19001，见 README 故障排查），Java API 必须显式 set。

**FileExists.java** —— 功能 (1)：上传，已存在时用户选择追加/覆盖
```java
package sdu.bigdata;

import org.apache.hadoop.fs.*;
import java.io.*;
import java.util.Scanner;

public class FileExists {
    public static void main(String[] args) throws Exception {
        FileSystem fs = HdfsApi.getFs();
        Scanner sc = new Scanner(System.in);
        System.out.print("本地路径: ");  String local = sc.next();
        System.out.print("HDFS 路径: "); String hdfs  = sc.next();
        Path dst = new Path(hdfs);
        if (fs.exists(dst)) {
            System.out.print("文件已存在，追加(a) / 覆盖(c)? ");
            String op = sc.next();
            if (op.equalsIgnoreCase("a")) {
                FSDataOutputStream out = fs.append(dst);
                InputStream in = new BufferedInputStream(new FileInputStream(local));
                byte[] buf = new byte[4096]; int n;
                while ((n = in.read(buf)) > 0) out.write(buf, 0, n);
                in.close(); out.close();
            } else {
                fs.copyFromLocalFile(false, true, new Path(local), dst); // overwrite=true
            }
        } else {
            fs.copyFromLocalFile(new Path(local), dst);
        }
        System.out.println("完成，当前内容:");
        FSDataInputStream in = fs.open(dst);
        org.apache.hadoop.io.IOUtils.copyBytes(in, System.out, 4096, false);
        fs.close();
    }
}
```

**LSFile.java** —— 功能 (4)(5)：列出文件/递归目录信息
```java
package sdu.bigdata;

import org.apache.hadoop.fs.*;

public class LSFile {
    public static void main(String[] args) throws Exception {
        FileSystem fs = HdfsApi.getFs();
        list(fs, new Path(args.length > 0 ? args[0] : "/user/gray/local.txt"),
             args.length > 1 && args[1].equals("R"));
        fs.close();
    }
    static void list(FileSystem fs, Path p, boolean recursive) throws Exception {
        FileStatus[] st = recursive ? fs.listStatus(p) : new FileStatus[]{ fs.getFileStatus(p) };
        for (FileStatus s : st) {
            System.out.printf("%s %d %s %s %d %tc %s%n",
                s.getPermission(), s.getReplication(), s.getOwner(), s.getGroup(),
                s.getLen(), s.getModificationTime(), s.getPath());
        }
    }
}
```

**AppendFile.java** —— 功能 (8)：追加到开头或结尾
```java
package sdu.bigdata;

import org.apache.hadoop.fs.*;
import java.io.*;
import java.util.Scanner;

public class AppendFile {
    public static void main(String[] args) throws Exception {
        FileSystem fs = HdfsApi.getFs();
        Scanner sc = new Scanner(System.in);
        System.out.print("本地追加内容文件: "); String local = sc.next();
        System.out.print("HDFS 目标文件: ");   String hdfs  = sc.next();
        System.out.print("追加到开头(b) / 结尾(e)? "); String op = sc.next();
        Path dst = new Path(hdfs);
        if (op.equalsIgnoreCase("e")) {              // 结尾：直接 append
            FSDataOutputStream out = fs.append(dst);
            byte[] buf = java.nio.file.Files.readAllBytes(new File(local).toPath());
            out.write(buf); out.close();
        } else {                                     // 开头：读原文件 -> 新文件 = 追加内容 + 原内容
            FSDataInputStream in = fs.open(dst);
            File tmp = File.createTempFile("hdfs", ".tmp");
            tmp.deleteOnExit();
            java.nio.file.Files.copy(in, tmp.toPath());
            in.close();
            FSDataOutputStream out = fs.create(dst, true);
            out.write(java.nio.file.Files.readAllBytes(new File(local).toPath()));
            out.write(java.nio.file.Files.readAllBytes(tmp.toPath()));
            out.close();
        }
        System.out.println("追加后内容:");
        FSDataInputStream in2 = fs.open(dst);
        org.apache.hadoop.io.IOUtils.copyBytes(in2, System.out, 4096, false);
        fs.close();
    }
}
```

## 3.1 合并去重（MergeDedup）

**思路**：Map 输出 key=整行文本，value 空即可（用 NullWritable 或行内容本身）；_shuffle_ 天然按 key 分组，同 key 只进同一个 reduce；Reduce 只输出一次 key，即天然去重；MapReduce 默认按键排序，输出自动有序 —— "合并 + 去重 + 排序"一步到位。

**MergeDedup.java**
```java
package sdu.bigdata;

import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.*;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;

public class MergeDedup {
    public static class M extends Mapper<Object, Text, Text, Text> {
        public void map(Object k, Text v, Context c) throws Exception {
            String line = v.toString().trim();
            if (!line.isEmpty()) c.write(new Text(line), new Text(""));
        }
    }
    public static class R extends Reducer<Text, Text, Text, Text> {
        public void reduce(Text k, Iterable<Text> vs, Context c) throws Exception {
            c.write(k, new Text(""));    // 同一 key 只输出一次 → 去重
        }
    }
    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        Job job = Job.getInstance(conf, "merge dedup");
        job.setJarByClass(MergeDedup.class);
        job.setMapperClass(M.class);
        job.setReducerClass(R.class);
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(Text.class);
        FileInputFormat.addInputPath(job, new Path(args[0]));
        FileOutputFormat.setOutputPath(job, new Path(args[1]));
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}
```

## 3.2 数字排序（SortInts）

**思路**：Map 把每个整数读成 key，value 放原文件路径/序号；MapReduce 框架对 key 的默认排序是**数值型 Integer 键按数值升序**（注意：Text 类型是按字典序，`"10" < "9"` 会错，所以必须用 `IntWritable` 作 key）；Reduce 里用一个全局计数器（自定义 `Counter` 或静态计数器在 setup 中初始化）输出 `位次 数字`。
> 框架只会对 key 排序，因此 key 必须是整数本身，位次在 reduce 端生成。

**SortInts.java**
```java
package sdu.bigdata;

import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.*;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;
import java.io.IOException;

public class SortInts {
    public static class M extends Mapper<Object, Text, IntWritable, IntWritable> {
        public void map(Object k, Text v, Context c) throws IOException, InterruptedException {
            String s = v.toString().trim();
            if (!s.isEmpty()) c.write(new IntWritable(Integer.parseInt(s)), new IntWritable(1));
        }
    }
    public static class R extends Reducer<IntWritable, IntWritable, IntWritable, IntWritable> {
        private int rank = 0;
        public void setup(Context c) { rank = 0; }        // 每个 reduce 任务开始时清零
        public void reduce(IntWritable k, Iterable<IntWritable> vs, Context c) throws IOException, InterruptedException {
            rank++;                                       // 相同数字只占一个名次（若需每个重复数字各占名次，改为循环输出）
            c.write(new IntWritable(rank), k);
        }
    }
    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        Job job = Job.getInstance(conf, "sort ints");
        job.setJarByClass(SortInts.class);
        job.setMapperClass(M.class);
        job.setReducerClass(R.class);
        job.setMapOutputKeyClass(IntWritable.class);
        job.setMapOutputValueClass(IntWritable.class);
        job.setOutputKeyClass(IntWritable.class);
        job.setOutputValueClass(IntWritable.class);
        FileInputFormat.addInputPath(job, new Path(args[0]));
        FileOutputFormat.setOutputPath(job, new Path(args[1]));
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}
```
> ⚠️ 单个 reduce 时（`job.setNumReduceTasks(1)`，默认就是 1）位次全局连续；若 reduce 个数 >1，每个 reduce 计数器独立，位次会分段重复 —— 本题保持默认 1 个 reduce 即可。

**运行**
```bash

printf '33\n37\n12\n40\n'  > /tmp/mr_exp/sort/f1.txt
printf '4\n16\n39\n5\n'    > /tmp/mr_exp/sort/f2.txt
printf '1\n45\n25\n'       > /tmp/mr_exp/sort/f3.txt
hdfs dfs -mkdir -p /input/sort && hdfs dfs -put -f /tmp/mr_exp/sort/f1.txt /tmp/mr_exp/sort/f2.txt /tmp/mr_exp/sort/f3.txt /input/sort
hadoop jar target/mapreduce-exp-1.0.jar sdu.bigdata.SortInts /input/sort /output/sort
hdfs dfs -cat /output/sort/part-r-00000
```
✅ 输出应为 `1 1 / 2 4 / 3 5 / ... / 11 45`，与样例一致。

## 3.3 祖孙关系挖掘（Grandparent）

**思路**：经典的"连接"问题。child-parent 表要"自己和自己连接"：对每行 `(child, parent)`，Map 阶段发两条记录：
- key=`parent`，value=`c:child`（我是别人的 parent，等待找我的孙辈）
- key=`child`，value=`p:parent`（我是别人的 child，等待找我的祖辈）

同一个 key（同一个人）会在同一个 reduce 里汇合：作为 parent 收到的 `c:` 项（他的孩子）× 作为 child 收到的 `p:` 项（他的父母）两两组合，输出 `(grandchild, grandparent)`。

**Grandparent.java**
```java
package sdu.bigdata;

import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.*;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;
import java.io.IOException;
import java.util.*;

public class Grandparent {
    public static class M extends Mapper<Object, Text, Text, Text> {
        public void map(Object k, Text v, Context c) throws IOException, InterruptedException {
            String[] t = v.toString().trim().split("\\s+");
            if (t.length != 2 || t[0].equals("child")) return;  // 跳过表头
            c.write(new Text(t[1]), new Text("c:" + t[0]));     // key=parent: 候选孙辈
            c.write(new Text(t[0]), new Text("p:" + t[1]));     // key=child: 候选祖辈
        }
    }
    public static class R extends Reducer<Text, Text, Text, Text> {
        public void reduce(Text key, Iterable<Text> vals, Context c) throws IOException, InterruptedException {
            List<String> grandChildren = new ArrayList<>(), grandParents = new ArrayList<>();
            for (Text t : vals) {
                String s = t.toString();
                if (s.startsWith("c:")) grandChildren.add(s.substring(2));
                else grandParents.add(s.substring(2));
            }
            for (String gc : grandChildren)
                for (String gp : grandParents)
                    c.write(new Text(gc), new Text(gp));
        }
    }
    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        Job job = Job.getInstance(conf, "grandparent");
        job.setJarByClass(Grandparent.class);
        job.setMapperClass(M.class);
        job.setReducerClass(R.class);
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(Text.class);
        FileInputFormat.addInputPath(job, new Path(args[0]));
        FileOutputFormat.setOutputPath(job, new Path(args[1]));
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}
```
✅ 输出 12 行 `grandchild<TAB>grandparent`，与样例一致（Steven/Jone×{Alice,Jesse,Mary,Frank}，Philip/Mark×{Alice,Jesse}）。

## 常见报错速查
| 报错 | 解决 |
|---|---|
| `Output directory already exists` | `hdfs dfs -rm -r /output/xxx` 后重跑 |
| `Name node is in safe mode` | 等半分钟或 `hdfs dfsadmin -safemode leave` |
| `Connection refused`/`EOFException`（端口在监听仍失败） | WSL virtioproxy 回环转发失效，换 `fs.defaultFS` 端口（9000→19000→19001 曾先后生效），Java API 同步改 HdfsApi.java |
| Java API `Connection refused 9000` | 代码里必须 `conf.set("fs.defaultFS","hdfs://localhost:19001")` |
| MapReduce 输出 key 是 Text 时 `"10"<"9"` | 排序数字必须用 `IntWritable` 作 key |
| `mvn` 下载依赖慢/失败 | 配置阿里云 Maven 镜像或重试 |
| 抓取/下载超时 | 部分站点被墙，换用可直连源 |
